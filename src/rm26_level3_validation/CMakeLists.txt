cmake_minimum_required(VERSION 3.20)
project(rm26_level3_validation VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(tech_core_planner REQUIRED)
find_package(rm26_engineering_arm_demo REQUIRED)
find_package(rm27_heavy_arm_demo REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(PLANNER_PREFIX "${tech_core_planner_DIR}/../../.." ABSOLUTE)
add_executable(level3_case src/level3_case.cpp)
target_compile_definitions(level3_case PRIVATE LEVEL3_ASSET_DIR="${PLANNER_PREFIX}/${CMAKE_INSTALL_DATADIR}/tech_core_planner/models")
find_package(glfw3 REQUIRED)
find_package(OpenGL REQUIRED)
set(VIEWER_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../rm_tech_core_trajectory_solve/src/tech_core_planner/src")
target_sources(level3_case PRIVATE
  "${VIEWER_SOURCE_DIR}/visualization/playback.cpp"
  "${VIEWER_SOURCE_DIR}/visualization/viewer.cpp")
target_include_directories(level3_case PRIVATE "${VIEWER_SOURCE_DIR}")
target_link_libraries(level3_case PRIVATE glfw OpenGL::GL)
target_compile_features(level3_case PRIVATE cxx_std_17)
target_link_libraries(level3_case PRIVATE rm26_engineering_arm_demo::rm26_engineering_arm_demo rm27_heavy_arm_demo::rm27_heavy_arm_demo MuJoCo::MuJoCo)
target_compile_options(level3_case PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(arm_dimensions_fk src/arm_dimensions_fk.cpp)
target_compile_features(arm_dimensions_fk PRIVATE cxx_std_17)
target_link_libraries(arm_dimensions_fk PRIVATE rm27_heavy_arm_demo::rm27_heavy_arm_demo MuJoCo::MuJoCo)
target_compile_options(arm_dimensions_fk PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS level3_case arm_dimensions_fk RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS src/level3_batch.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME level3_batch)
install(PROGRAMS src/arm_dimensions_gui.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME arm_dimensions_gui)
install(PROGRAMS src/rm27_length_search.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME rm27_length_search)
install(PROGRAMS src/rm27_short_length_search.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME rm27_short_length_search)
install(FILES src/camera_visibility.py src/arm_variants.py DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY config DESTINATION ${CMAKE_INSTALL_DATADIR}/${PROJECT_NAME})
install(FILES package.xml README.md README-en.md DESTINATION ${CMAKE_INSTALL_DATADIR}/${PROJECT_NAME})
